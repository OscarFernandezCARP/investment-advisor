const express = require('express'); const router = express.Router(); const fs = require('fs'); const path = require('path'); const PORTFOLIO_PATH = path.join(__dirname, '../data/portfolio.json'); const TWELVE_KEY = process.env.TWELVE_DATA_KEY || 'ac2af7f74a4744cc98a1dc6534f112eb'; let cache = null, cacheTime = null; router.get('/', async (req, res) => { try { if (cache && (Date.now() - cacheTime) < 5 * 60 * 1000) { return res.json({ success: true, data: cache, cached: true }); } const portfolio = JSON.parse(fs.readFileSync(PORTFOLIO_PATH, 'utf8')); const assets = portfolio.assets; const symList = assets.map(a => a.symbol.replace('.BA','')).join(','); const url = `https://api.twelvedata.com/quote?symbol=${symList}&apikey=${TWELVE_KEY}`; const response = await fetch(url); const data = await response.json(); const result = assets.map(a => { const sym = a.symbol.replace('.BA',''); const q = data[sym] || {}; const price = parseFloat(q.close || 0); const prev = parseFloat(q.previous_close || price); const change = price - prev; const changePct = prev > 0 ? (change / prev) * 100 : 0; return { ...a, currentPrice: price || null, change: parseFloat(change.toFixed(2)), changePercent: parseFloat(changePct.toFixed(2)), currency: a.symbol.includes('.BA') ? 'ARS' : 'USD', totalValue: (a.quantity || 0) * price, gainLoss: a.purchasePrice > 0 ? ((price - a.purchasePrice) / a.purchasePrice) * 100 : 0, error: price === 0 ? 'Sin datos' : null }; }); cache = result; cacheTime = Date.now(); res.json({ success: true, data: result, cached: false }); } catch(e) { console.error(e); res.status(500).json({ success: false, error: e.message }); } }); router.get('/search/:symbol', async (req, res) => { try { const url = `https://api.twelvedata.com/quote?symbol=${req.params.symbol}&apikey=${TWELVE_KEY}`; const r = await fetch(url); const d = await r.json(); if (d.status === 'error') return res.status(404).json({ success: false, error: 'Simbolo no encontrado' }); res.json({ success: true, data: { symbol: d.symbol, name: d.name, price: parseFloat(d.close || 0), change: parseFloat(d.change || 0), changePercent: parseFloat(d.percent_change || 0), currency: d.currency, exchange: d.exchange, high: d.high, low: d.low, volume: d.volume }}); } catch(e) { res.status(500).json({ success: false, error: e.message }); } }); router.post('/update-holdings', (req, res) => { const { symbol, quantity, purchasePrice } = req.body; try { const portfolio = JSON.parse(fs.readFileSync(PORTFOLIO_PATH, 'utf8')); const idx = portfolio.assets.findIndex(a => a.symbol === symbol); if (idx === -1) return res.status(404).json({ success: false, error: 'No encontrado' }); portfolio.assets[idx].quantity = parseFloat(quantity) || 0; portfolio.assets[idx].purchasePrice = parseFloat(purchasePrice) || 0; fs.writeFileSync(PORTFOLIO_PATH, JSON.stringify(portfolio, null, 2)); cache = null; res.json({ success: true, message: symbol + ' actualizado' }); } catch(e) { res.status(500).json({ success: false, error: e.message }); } }); module.exports = router;
